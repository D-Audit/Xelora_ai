# bcrypt is pinned to 4.0.1 deliberately: passlib 1.7.4 (the last
# release) is incompatible with bcrypt>=4.1 and raises
# "password cannot be longer than 72 bytes" on every hash attempt,
# even for short passwords, due to a passlib bug in its own internal
# version-detection routine. Pinning bcrypt avoids it.

bcrypt==4.0.1
python-jose[cryptography]==3.3.0
